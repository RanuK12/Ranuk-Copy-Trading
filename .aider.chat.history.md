def load_historical_data(start_date: str, end_date: str) -> List[MarketFrame]:
    """

    logger.info(f"Loading historical data from {start_date} to {end_date}")
    
    # Parse dates

    # Initialize cache
    cache = {}
    
    # Initialize frames
    frames = []
    
    # Loop through each day
    current_date = start
